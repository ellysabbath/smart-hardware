function Protected({ children, roles }: { children: React.ReactNode; roles: Role[] }) {
  const { user } = useAuth()
  if (!user) return <Navigate to="/login" replace />
  if (!roles.includes(user.role)) return <Navigate to={defaultHome(user.role)} replace />
  return <>{children}</>
}

const defaultHome = (role: Role) =>
  role === 'admin' ? '/dashboard'
    : role === 'seller' ? '/pos'
      : role === 'warehouse' ? '/stock'
        : '/owner'

export default function App() {
  const { user } = useAuth()

  return (
    <Routes>
      <Route path="/login" element={user ? <Navigate to={defaultHome(user.role)} replace /> : <Login />} />

      {/* Admin */}
      <Route path="/dashboard" element={<Protected roles={['admin']}><AdminDashboard /></Protected>} />
      <Route path="/users" element={<Protected roles={['admin']}><UsersPage /></Protected>} />
      <Route path="/audit" element={<Protected roles={['admin']}><AuditPage /></Protected>} />
      <Route path="/settings" element={<Protected roles={['admin']}><SettingsPage /></Protected>} />

      {/* Seller */}
      <Route path="/pos" element={<Protected roles={['seller', 'admin']}><SellerPOS /></Protected>} />
      <Route path="/quotations" element={<Protected roles={['seller', 'admin']}><QuotationsPage /></Protected>} />
      <Route path="/sales" element={<Protected roles={['seller', 'admin', 'owner']}><SalesHistoryPage /></Protected>} />

      {/* Warehouse */}
      <Route path="/stock" element={<Protected roles={['warehouse', 'admin']}><StockPage /></Protected>} />
      <Route path="/grn" element={<Protected roles={['warehouse', 'admin']}><GRNPage /></Protected>} />
      <Route path="/stock-take" element={<Protected roles={['warehouse', 'admin']}><StockTakePage /></Protected>} />

      {/* Owner */}
      <Route path="/owner" element={<Protected roles={['owner', 'admin']}><OwnerDashboard /></Protected>} />
      <Route path="/profit-loss" element={<Protected roles={['owner', 'admin']}><ProfitLossPage /></Protected>} />
      <Route path="/reports" element={<Protected roles={['owner', 'admin']}><ReportsPage /></Protected>} />

      {/* Shared */}
      <Route path="/products" element={<Protected roles={['admin', 'seller', 'warehouse', 'owner']}><ProductsPage /></Protected>} />
      <Route path="/purchases" element={<Protected roles={['admin', 'warehouse']}><PurchasesPage /></Protected>} />
      <Route path="/customers" element={<Protected roles={['admin', 'seller']}><CustomersPage /></Protected>} />
      <Route path="/suppliers" element={<Protected roles={['admin', 'warehouse']}><SuppliersPage /></Protected>} />
      <Route path="/expenses" element={<Protected roles={['admin', 'owner', 'seller']}><ExpensesPage /></Protected>} />
      <Route path="/zakat" element={<Protected roles={['admin', 'owner']}><ZakatPage /></Protected>} />

      {/* Fallback */}
      <Route path="*" element={<Navigate to={user ? defaultHome(user.role) : '/login'} replace />} />
    </Routes>
  )
}

import { useState } from 'react'
import Layout from '../../components/Layout'
import { useData } from '../../context/DataContext'
import Button from '../../components/ui/Button'
import Modal from '../../components/ui/Modal'
import { Table, Td } from '../../components/ui/Table'
import Badge from '../../components/ui/Badge'
import { Card } from '../../components/ui/Card'
import { fmtTsh } from '../../lib/utils'
import { ClipboardCheck, AlertTriangle } from 'lucide-react'

export default function StockTakePage() {
  const { products, adjustStock } = useData()
  const [counts, setCounts] = useState<Record<string, number>>({})
  const [confirmOpen, setConfirmOpen] = useState(false)
  const [submitted, setSubmitted] = useState(false)

  const diffs = products.map(p => {
    const counted = counts[p.id] ?? p.stock
    return { p, counted, diff: counted - p.stock }
  })
  const withDiffs = diffs.filter(d => d.diff !== 0)
  const totalShortage = withDiffs.filter(d => d.diff < 0).reduce((s, d) => s + Math.abs(d.diff) * d.p.buyPrice, 0)

  const apply = () => {
    withDiffs.forEach(d => {
      adjustStock(d.p.id, d.diff, 'adjust', `Hesabu ya ghala: ${d.p.stock} → ${d.counted}`)
    })
    setSubmitted(true)
    setConfirmOpen(false)
    setCounts({})
  }

  return (
    <Layout title="Hesabu ya Ghala (Stock Take)">
      <div className="mb-5 bg-blue-50 border border-blue-200 rounded-lg p-3 text-sm text-blue-800">
        Hesabu kimwili, kisha ingiza idadi halisi kwenye jedwali. System itaonyesha tofauti (wizi / upotevu).
      </div>

      {submitted && (
        <div className="mb-4 bg-emerald-50 border border-emerald-200 rounded-lg p-3 text-sm text-emerald-700">
          ✓ Hesabu imehifadhiwa. Stoo imesasishwa.
        </div>
      )}

      <Card className="mb-5 p-4">
        <div className="flex items-center gap-3">
          <div className="flex-1">
            <div className="text-xs text-slate-500">Bidhaa zenye tofauti</div>
            <div className="text-2xl font-bold">{withDiffs.length}</div>
          </div>
          <div className="flex-1">
            <div className="text-xs text-slate-500">Thamani ya upungufu</div>
            <div className="text-2xl font-bold text-red-600">{fmtTsh(totalShortage)}</div>
          </div>
          <Button onClick={() => setConfirmOpen(true)} disabled={withDiffs.length === 0}>
            <span className="flex items-center gap-2"><ClipboardCheck size={16} /> Hifadhi Hesabu</span>
          </Button>
        </div>
      </Card>

      <Card>
        <Table head={['Bidhaa', 'SKU', 'Stoo ya System', 'Idadi Halisi', 'Tofauti', 'Thamani']}>
          {diffs.map(d => (
            <tr key={d.p.id} className={d.diff !== 0 ? 'bg-amber-50' : ''}>
              <Td className="font-medium">{d.p.name}</Td>
              <Td className="font-mono text-xs">{d.p.sku}</Td>
              <Td>{d.p.stock} {d.p.unit}</Td>
              <Td>
                <input type="number" min={0} value={counts[d.p.id] ?? d.p.stock}
                  onChange={e => setCounts({ ...counts, [d.p.id]: Number(e.target.value) || 0 })}
                  className="w-24 border rounded px-2 py-1 text-sm" />
              </Td>
              <Td>
                {d.diff === 0 ? <span className="text-slate-400">—</span>
                  : d.diff < 0 ? <Badge color="red">Upungufu {d.diff}</Badge>
                  : <Badge color="green">Ziada +{d.diff}</Badge>}
              </Td>
              <Td>{d.diff !== 0 ? fmtTsh(Math.abs(d.diff) * d.p.buyPrice) : '—'}</Td>
            </tr>
          ))}
        </Table>
      </Card>

      <Modal open={confirmOpen} onClose={() => setConfirmOpen(false)} title="Thibitisha Hesabu" size="sm">
        <div className="space-y-3">
          <div className="flex items-start gap-2 bg-amber-50 border border-amber-200 rounded-lg p-3">
            <AlertTriangle className="text-amber-600 flex-shrink-0" size={20} />
            <div className="text-sm">
              <p className="font-medium">Bidhaa {withDiffs.length} zina tofauti.</p>
              <p className="text-slate-600 mt-1">Stoo itasasishwa kulingana na hesabu halisi. Kitendo hiki hakiwezi kurudishwa.</p>
            </div>
          </div>
          <div className="flex justify-end gap-3">
            <Button variant="outline" onClick={() => setConfirmOpen(false)}>Ghairi</Button>
            <Button variant="success" onClick={apply}>Thibitisha</Button>
          </div>
        </div>
      </Modal>
    </Layout>
  )
}

import { useState } from 'react'
import Layout from '../../components/Layout'
import { useData } from '../../context/DataContext'
import Button from '../../components/ui/Button'
import Input from '../../components/ui/Input'
import Modal from '../../components/ui/Modal'
import { Table, Td } from '../../components/ui/Table'
import Badge from '../../components/ui/Badge'
import { Card, StatCard } from '../../components/ui/Card'
import { fmtTsh, daysBetween, today } from '../../lib/utils'
import { AlertTriangle, Package, Warehouse, ArrowRightLeft, Trash2 } from 'lucide-react'

export default function StockPage() {
  const { products, stockMovements, adjustStock } = useData()
  const [tab, setTab] = useState<'stock' | 'movements'>('stock')
  const [transferOpen, setTransferOpen] = useState(false)
  const [damageOpen, setDamageOpen] = useState(false)
  const [productId, setProductId] = useState('')
  const [qty, setQty] = useState(1)
  const [fromLoc, setFromLoc] = useState('')
  const [toLoc, setToLoc] = useState('')
  const [reason, setReason] = useState('')

  const lowStock = products.filter(p => p.stock <= p.reorderLevel)
  const expiring = products.filter(p => p.expiryDate && daysBetween(today(), p.expiryDate) <= 30)

  const submitTransfer = () => {
    if (!productId || qty <= 0) return
    adjustStock(productId, -qty, 'transfer', `Transfer: ${fromLoc} → ${toLoc}`, toLoc)
    setTransferOpen(false)
    setProductId(''); setQty(1); setFromLoc(''); setToLoc('')
  }

  const submitDamage = () => {
    if (!productId || qty <= 0) return
    adjustStock(productId, -qty, 'damage', reason || 'Uharibifu', undefined)
    setDamageOpen(false)
    setProductId(''); setQty(1); setReason('')
  }

  return (
    <Layout title="Usimamizi wa Stoo">
      <div className="grid grid-cols-2 md:grid-cols-4 gap-3 mb-5">
        <StatCard label="Jumla ya Bidhaa" value={String(products.length)} icon={<Package />} />
        <StatCard label="Chini ya Kiwango" value={String(lowStock.length)} icon={<AlertTriangle />} color="amber" />
        <StatCard label="Zinaisha (30d)" value={String(expiring.length)} icon={<AlertTriangle />} color="red" />
        <StatCard label="Thamani Stoo" value={fmtTsh(products.reduce((s, p) => s + p.stock * p.buyPrice, 0))} icon={<Warehouse />} color="green" />
      </div>

      <div className="flex gap-2 mb-4">
        <button onClick={() => setTab('stock')}
          className={`px-4 py-2 text-sm rounded-lg ${tab === 'stock' ? 'bg-brand-600 text-white' : 'bg-white border border-slate-200 text-slate-700'}`}>
          Stoo ya Sasa
        </button>
        <button onClick={() => setTab('movements')}
          className={`px-4 py-2 text-sm rounded-lg ${tab === 'movements' ? 'bg-brand-600 text-white' : 'bg-white border border-slate-200 text-slate-700'}`}>
          Historia ya Mwendo
        </button>
        <div className="flex-1" />
        <Button variant="outline" onClick={() => { setProductId(''); setQty(1); setFromLoc(''); setToLoc(''); setTransferOpen(true) }}>
          <span className="flex items-center gap-2"><ArrowRightLeft size={16} /> Hamisha</span>
        </Button>
        <Button variant="danger" onClick={() => { setProductId(''); setQty(1); setReason(''); setDamageOpen(true) }}>
          <span className="flex items-center gap-2"><Trash2 size={16} /> Rekodi Uharibifu</span>
        </Button>
      </div>

      {tab === 'stock' && (
        <Card>
          <Table head={['Bidhaa', 'SKU', 'Aina', 'Stoo', 'Mahali', 'Kiwango', 'Hali']}>
            {products.map(p => (
              <tr key={p.id} className="hover:bg-slate-50">
                <Td className="font-medium">{p.name}</Td>
                <Td className="font-mono text-xs">{p.sku}</Td>
                <Td><Badge color="blue">{p.category}</Badge></Td>
                <Td><b>{p.stock}</b> {p.unit}</Td>
                <Td>{p.location || '—'}</Td>
                <Td>{p.reorderLevel}</Td>
                <Td>
                  {p.stock === 0 ? <Badge color="red">Hakuna</Badge>
                    : p.stock <= p.reorderLevel ? <Badge color="amber">Chini</Badge>
                    : <Badge color="green">Sawa</Badge>}
                </Td>
              </tr>
            ))}
          </Table>
        </Card>
      )}

      {tab === 'movements' && (
        <Card>
          <Table head={['Tarehe', 'Bidhaa', 'Aina', 'Qty', 'Kutoka', 'Kwenda', 'Maelezo', 'Aliyeingiza']}>
            {stockMovements.slice(0, 100).map(m => (
              <tr key={m.id} className="hover:bg-slate-50">
                <Td className="text-xs">{m.createdAt.slice(0, 16).replace('T', ' ')}</Td>
                <Td>{m.productName}</Td>
                <Td><Badge color={
                  m.type === 'sale' ? 'blue' : m.type === 'purchase' ? 'green'
                    : m.type === 'damage' ? 'red' : m.type === 'transfer' ? 'purple' : 'slate'
                }>{m.type}</Badge></Td>
                <Td className={m.qty < 0 ? 'text-red-600' : 'text-emerald-600'}><b>{m.qty > 0 ? '+' : ''}{m.qty}</b></Td>
                <Td>{m.fromLocation || '—'}</Td>
                <Td>{m.toLocation || '—'}</Td>
                <Td className="text-xs">{m.note || '—'}</Td>
                <Td className="text-xs">{m.createdBy}</Td>
              </tr>
            ))}
            {stockMovements.length === 0 && (
              <tr><td colSpan={8} className="text-center py-10 text-slate-500">Hakuna historia</td></tr>
            )}
          </Table>
        </Card>
      )}

      {/* TRANSFER MODAL */}
      <Modal open={transferOpen} onClose={() => setTransferOpen(false)} title="Hamisha Stoki" size="md">
        <div className="space-y-4">
          <select value={productId} onChange={e => setProductId(e.target.value)}
            className="w-full border rounded-lg px-3 py-2 text-sm">
            <option value="">— Chagua bidhaa —</option>
            {products.map(p => <option key={p.id} value={p.id}>{p.name} ({p.stock} {p.unit})</option>)}
          </select>
          <Input label="Idadi" type="number" value={qty} onChange={e => setQty(Number(e.target.value))} />
          <Input label="Kutoka (Location)" value={fromLoc} onChange={e => setFromLoc(e.target.value)} placeholder="Mfano: Kwembe" />
          <Input label="Kwenda (Location)" value={toLoc} onChange={e => setToLoc(e.target.value)} placeholder="Mfano: Pugu" />
          <div className="flex justify-end gap-3">
            <Button variant="outline" onClick={() => setTransferOpen(false)}>Ghairi</Button>
            <Button onClick={submitTransfer}>Hamisha</Button>
          </div>
        </div>
      </Modal>

      {/* DAMAGE MODAL */}
      <Modal open={damageOpen} onClose={() => setDamageOpen(false)} title="Rekodi Uharibifu / Upotevu" size="md">
        <div className="space-y-4">
          <select value={productId} onChange={e => setProductId(e.target.value)}
            className="w-full border rounded-lg px-3 py-2 text-sm">
            <option value="">— Chagua bidhaa —</option>
            {products.map(p => <option key={p.id} value={p.id}>{p.name} ({p.stock} {p.unit})</option>)}
          </select>
          <Input label="Idadi iliyoharibika" type="number" value={qty} onChange={e => setQty(Number(e.target.value))} />
          <Input label="Sababu / Maelezo" value={reason} onChange={e => setReason(e.target.value)}
            placeholder="Mfano: Saruji imekauka, mbao ziliwa na mchwa" />
          <div className="flex justify-end gap-3">
            <Button variant="outline" onClick={() => setDamageOpen(false)}>Ghairi</Button>
            <Button variant="danger" onClick={submitDamage}>Rekodi Hasara</Button>
          </div>
        </div>
      </Modal>
    </Layout>
  )
}
import { useState } from 'react'
import Layout from '../../components/Layout'
import { useData } from '../../context/DataContext'
import Button from '../../components/ui/Button'
import Modal from '../../components/ui/Modal'
import { Table, Td } from '../../components/ui/Table'
import Badge from '../../components/ui/Badge'
import { Card } from '../../components/ui/Card'
import type { Purchase } from '../../types'
import { fmtTsh } from '../../lib/utils'
import { Package, AlertCircle } from 'lucide-react'

export default function GRNPage() {
  const { purchases, grns, receivePurchase } = useData()
  const [open, setOpen] = useState(false)
  const [selected, setSelected] = useState<Purchase | null>(null)
  const [received, setReceived] = useState<Record<string, number>>({})

  const pending = purchases.filter(p => p.status !== 'received')

  const startReceiving = (p: Purchase) => {
    setSelected(p)
    setReceived(Object.fromEntries(p.items.map(i => [i.productId, i.qty])))
    setOpen(true)
  }

  const submit = () => {
    if (!selected) return
    const items = Object.entries(received).map(([productId, qty]) => ({ productId, qty }))
    receivePurchase(selected.id, items)
    setOpen(false)
    setSelected(null)
  }

  return (
    <Layout title="Kupokea Mizigo (GRN)">
      <div className="mb-5 text-sm text-slate-500">
        Rekodi mizigo iliyopokelewa dhidi ya LPO. System itaonyesha upungufu.
      </div>

      <Card className="mb-5">
        <div className="p-4 border-b border-slate-200 font-semibold">Oda Zinazosubiri Kupokelewa</div>
        <Table head={['LPO', 'Msambazaji', 'Jumla', 'Hali', 'Tarehe', '']}>
          {pending.map(p => (
            <tr key={p.id} className="hover:bg-slate-50">
              <Td className="font-mono text-xs">{p.lpoNo}</Td>
              <Td>{p.supplierName}</Td>
              <Td className="font-medium">{fmtTsh(p.total)}</Td>
              <Td><Badge color={p.status === 'partial' ? 'amber' : 'blue'}>{p.status}</Badge></Td>
              <Td className="text-xs">{p.createdAt.slice(0, 10)}</Td>
              <Td>
                <Button size="sm" onClick={() => startReceiving(p)}>
                  <span className="flex items-center gap-1"><Package size={14} /> Pokea</span>
                </Button>
              </Td>
            </tr>
          ))}
          {pending.length === 0 && (
            <tr><td colSpan={6} className="text-center py-10 text-slate-500">Hakuna oda zinazosubiri</td></tr>
          )}
        </Table>
      </Card>

      <Card>
        <div className="p-4 border-b border-slate-200 font-semibold">Historia ya GRN</div>
        <Table head={['GRN', 'LPO', 'Tarehe', 'Bidhaa Zilizopokelewa', '']}>
          {grns.map(g => {
            const purch = purchases.find(p => p.id === g.purchaseId)
            const shortages = g.items.filter(i => i.received < i.ordered)
            return (
              <tr key={g.id} className="hover:bg-slate-50">
                <Td className="font-mono text-xs">{g.grnNo}</Td>
                <Td className="font-mono text-xs">{purch?.lpoNo || '—'}</Td>
                <Td className="text-xs">{g.createdAt.slice(0, 16).replace('T', ' ')}</Td>
                <Td>{g.items.map(i => `${i.productName}: ${i.received}/${i.ordered}`).join(', ')}</Td>
                <Td>
                  {shortages.length > 0 && (
                    <Badge color="red">
                      <span className="flex items-center gap-1"><AlertCircle size={10} /> Upungufu</span>
                    </Badge>
                  )}

import { useState } from 'react'

interface Line { productId: string; qty: number; price: number }

export default function PurchasesPage() {
  const { products, suppliers, purchases, addPurchase } = useData()
  const [open, setOpen] = useState(false)
  const [supplierId, setSupplierId] = useState('')
  const [lines, setLines] = useState<Line[]>([])
  const [productToAdd, setProductToAdd] = useState('')

  const linesDetailed = lines.map(l => {
    const p = products.find(x => x.id === l.productId)!
    return { ...l, product: p }
  }).filter(l => l.product)
  const total = linesDetailed.reduce((s, l) => s + l.price * l.qty, 0)

  const addLine = () => {
    if (!productToAdd) return
    const p = products.find(x => x.id === productToAdd)!
    if (lines.some(l => l.productId === productToAdd)) {
      setLines(lines.map(l => l.productId === productToAdd ? { ...l, qty: l.qty + 1 } : l))
    } else {
      setLines([...lines, { productId: productToAdd, qty: 1, price: p.buyPrice }])
    }
    setProductToAdd('')
  }

  const save = () => {
    const sup = suppliers.find(s => s.id === supplierId)
    if (!sup || lines.length === 0) return
    const items: PurchaseItem[] = linesDetailed.map(l => ({
      productId: l.product.id, productName: l.product.name, qty: l.qty, price: l.price,
    }))
    addPurchase({ supplierId, supplierName: sup.name, items, total, status: 'pending' })
    setOpen(false); setSupplierId(''); setLines([])
  }

  return (
    <Layout title="Ununuzi (LPO)">
      <div className="flex justify-end mb-5">
        <Button onClick={() => { setSupplierId(''); setLines([]); setOpen(true) }}>
          <span className="flex items-center gap-2"><Plus size={16} /> LPO Mpya</span>
        </Button>
      </div>

      <Card>
        <Table head={['LPO', 'Msambazaji', 'Bidhaa', 'Jumla', 'Hali', 'Tarehe', '']}>
          {purchases.map(p => (
            <tr key={p.id} className="hover:bg-slate-50">
              <Td className="font-mono text-xs">{p.lpoNo}</Td>
              <Td>{p.supplierName}</Td>
              <Td>{p.items.length} bidhaa</Td>
              <Td className="font-medium">{fmtTsh(p.total)}</Td>
              <Td><Badge color={p.status === 'received' ? 'green' : p.status === 'partial' ? 'amber' : 'blue'}>{p.status}</Badge></Td>
              <Td className="text-xs">{p.createdAt.slice(0, 10)}</Td>
              <Td>
                <button className="text-xs text-brand-600 hover:underline" title="Angalia">
                  <Truck size={14} />
                </button>
              </Td>
            </tr>
          ))}
          {purchases.length === 0 && (
            <tr><td colSpan={7} className="text-center py-10 text-slate-500">Hakuna LPO bado</td></tr>
          )}
        </Table>
      </Card>

      <Modal open={open} onClose={() => setOpen(false)} title="Oda Mpya ya Ununuzi (LPO)" size="lg">
        <div className="space-y-4">
          <Select label="Msambazaji" value={supplierId} onChange={e => setSupplierId(e.target.value)}>
            <option value="">— Chagua —</option>
            {suppliers.map(s => <option key={s.id} value={s.id}>{s.name}</option>)}
          </Select>

          <div className="flex gap-2">
            <div className="flex-1">
              <Select value={productToAdd} onChange={e => setProductToAdd(e.target.value)}>
                <option value="">— Ongeza bidhaa —</option>
                {products.map(p => <option key={p.id} value={p.id}>{p.name}</option>)}
              </Select>
            </div>
            <Button onClick={addLine} disabled={!productToAdd}>Ongeza</Button>
          </div>

          {linesDetailed.length > 0 && (
            <table className="w-full text-sm border">
              <thead className="bg-slate-100">
                <tr>
                  <th className="text-left px-2 py-1">Bidhaa</th>
                  <th className="text-right px-2 py-1">Idadi</th>
                  <th className="text-right px-2 py-1">Bei</th>
                  <th className="text-right px-2 py-1">Jumla</th>
                  <th></th>
                </tr>
              </thead>
              <tbody>
                {linesDetailed.map(l => (
                  <tr key={l.productId} className="border-b">
                    <td className="px-2 py-1">{l.product.name}</td>
                    <td className="text-right px-2 py-1">
                      <input type="number" value={l.qty}
                        onChange={e => setLines(lines.map(x => x.productId === l.productId ? { ...x, qty: Number(e.target.value) || 1 } : x))}
                        className="w-16 border rounded px-2 py-1 text-right" />
                    </td>
                    <td className="text-right px-2 py-1">
                      <input type="number" value={l.price}
                        onChange={e => setLines(lines.map(x => x.productId === l.productId ? { ...x, price: Number(e.target.value) || 0 } : x))}
                        className="w-24 border rounded px-2 py-1 text-right" />
                    </td>
                    <td className="text-right px-2 py-1 font-medium">{fmtTsh(l.price * l.qty)}</td>
                    <td className="text-right px-2 py-1">
                      <button onClick={() => setLines(lines.filter(x => x.productId !== l.productId))}
                        className="text-red-600 p-1 hover:bg-red-50 rounded"><Trash2 size={14} /></button>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          )}

          <div className="text-right text-lg font-bold">Jumla: <span className="text-brand-600">{fmtTsh(total)}</span></div>

          <div className="flex justify-end gap-3">
            <Button variant="outline" onClick={() => setOpen(false)}>Ghairi</Button>
            <Button onClick={save} disabled={!supplierId || lines.length === 0}>Hifadhi LPO</Button>
          </div>
        </div>
      </Modal>
    </Layout>
  )
}

import { useState } from 'react'
import Layout from '../../components/Layout'
import { useData } from '../../context/DataContext'
import Button from '../../components/ui/Button'
import Modal from '../../components/ui/Modal'
import { Table, Td } from '../../components/ui/Table'
import Badge from '../../components/ui/Badge'
import { Card } from '../../components/ui/Card'
import { fmtTsh, today } from '../../lib/utils'
import { Heart, CheckCircle2, Sparkles, TrendingUp } from 'lucide-react'

export default function ZakatPage() {
  const { sales, expenses, zakat, addZakatRecord, markZakatPaid } = useData()
  const [open, setOpen] = useState(false)
  const [period, setPeriod] = useState<'daily' | 'weekly' | 'monthly'>('daily')
  const [viewId, setViewId] = useState<string | null>(null)

  // compute profit for period
  const computeProfit = (type: 'daily' | 'weekly' | 'monthly') => {
    const now = new Date()
    const cutoff = new Date()
    if (type === 'daily') cutoff.setHours(0, 0, 0, 0)
    else if (type === 'weekly') cutoff.setDate(now.getDate() - 7)
    else cutoff.setDate(now.getDate() - 30)

    const salesInPeriod = sales.filter(s => new Date(s.createdAt) >= cutoff)
    const revenue = salesInPeriod.reduce((sum, s) => sum + s.total, 0)
    const cogs = salesInPeriod.reduce((sum, s) =>
      sum + s.items.reduce((x, i) => x + i.buyPrice * i.qty, 0), 0)
    const opex = expenses.filter(e => e.status === 'approved' && new Date(e.date) >= cutoff)
      .reduce((sum, e) => sum + e.amount, 0)
    const netProfit = Math.max(0, revenue - cogs - opex)
    return { revenue, cogs, opex, netProfit }
  }

  const preview = computeProfit(period)

  const save = () => {
    const z = preview.netProfit * 0.10
    const s = preview.netProfit * 0.10
    addZakatRecord({
      date: today(),
      type: period,
      netProfit: preview.netProfit,
      zakat: z,
      sadaka: s,
      total: z + s,
      paid: false,
    })
    setOpen(false)
  }

  const totalOwed = zakat.filter(z => !z.paid).reduce((s, z) => s + z.total, 0)
  const totalPaid = zakat.filter(z => z.paid).reduce((s, z) => s + z.total, 0)

  return (
    <Layout title="Zaka & Sadaka">
      <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mb-6">
        <Card className="p-5 bg-gradient-to-br from-emerald-500 to-emerald-700 text-white">
          <div className="flex items-center gap-2 text-emerald-100 text-xs mb-1"><Heart size={14} /> Kwa Bwana - Haijalipwa</div>
          <div className="text-2xl font-bold">{fmtTsh(totalOwed)}</div>
        </Card>
        <Card className="p-5">
          <div className="flex items-center gap-2 text-slate-500 text-xs mb-1"><CheckCircle2 size={14} /> Zilizolipwa</div>
          <div className="text-2xl font-bold text-emerald-600">{fmtTsh(totalPaid)}</div>
        </Card>
        <Card className="p-5">
          <div className="flex items-center gap-2 text-slate-500 text-xs mb-1"><Sparkles size={14} /> Kanuni</div>
          <div className="text-sm text-slate-700">Zaka 10% + Sadaka 10% = 20% ya Faida Halisi</div>
        </Card>
      </div>

      <div className="flex justify-between items-center mb-4">
        <div className="text-sm text-slate-500">Hesabu ya Zaka na Sadaka kwa kila kipindi</div>
        <Button onClick={() => setOpen(true)}>
          <span className="flex items-center gap-2"><TrendingUp size={16} /> Hesabu Mpya</span>
        </Button>
      </div>

      <Card>
        <Table head={['Tarehe', 'Kipindi', 'Faida Halisi', 'Zaka 10%', 'Sadaka 10%', 'Jumla', 'Hali', '']}>
          {zakat.map(z => (
            <tr key={z.id} className="hover:bg-slate-50">
              <Td className="text-xs">{z.date}</Td>
              <Td><Badge color="blue">{z.type === 'daily' ? 'Siku' : z.type === 'weekly' ? 'Wiki' : 'Mwezi'}</Badge></Td>
              <Td>{fmtTsh(z.netProfit)}</Td>
              <Td className="text-emerald-700">{fmtTsh(z.zakat)}</Td>
              <Td className="text-emerald-700">{fmtTsh(z.sadaka)}</Td>
              <Td className="font-bold">{fmtTsh(z.total)}</Td>
              <Td><Badge color={z.paid ? 'green' : 'amber'}>{z.paid ? 'Imelipwa' : 'Inadaiwa'}</Badge></Td>
              <Td>
                {!z.paid ? (
                  <button onClick={() => markZakatPaid(z.id)} className="text-xs text-emerald-600 hover:underline">Nimelipa</button>
                ) : (
                  <span className="text-xs text-slate-400">{z.paidDate}</span>
                )}
              </Td>
            </tr>
          ))}
          {zakat.length === 0 && (
            <tr><td colSpan={8} className="text-center py-10 text-slate-500">Hakuna rekodi bado</td></tr>
          )}
        </Table>
      </Card>

      <Modal open={open} onClose={() => setOpen(false)} title="Hesabu Zaka & Sadaka" size="md">
        <div className="space-y-4">
          <select value={period} onChange={e => setPeriod(e.target.value as any)}

    const p = products.find(x => x.id === l.productId)!
    return { ...l, product: p }
  }).filter(l => l.product)

  const total = subtotal - discount

    setCustomerId(''); setName(''); setLines([]); setDiscount(0); setProductToAdd('')

    } else {
      setLines([...lines, { productId: productToAdd, qty: 1 }])
    }
    setProductToAdd('')

    if (lines.length === 0) return
    const customer = customers.find(c => c.id === customerId)

      productId: l.product.id, productName: l.product.name, qty: l.qty,

interface CartLine { productId: string; qty: number; unit: string }

export default function SellerPOS() {
  const { user } = useAuth()
  const { products, customers, addSale, adjustStock } = useData()

  const [search, setSearch] = useState('')
  const [cart, setCart] = useState<CartLine[]>([])
  const [discount, setDiscount] = useState(0)
  const [customerId, setCustomerId] = useState('')
  const [walkInName, setWalkInName] = useState('')
  const [checkoutOpen, setCheckoutOpen] = useState(false)
  const [receipt, setReceipt] = useState<{ no: string; total: number; paid: number } | null>(null)

  // payments
  const [payCash, setPayCash] = useState(0)
  const [payMpesa, setPayMpesa] = useState(0)
  const [payTigo, setPayTigo] = useState(0)
  const [payBank, setPayBank] = useState(0)
  const [payCredit, setPayCredit] = useState(0)

  const filtered = useMemo(() => {
    const q = search.toLowerCase().trim()
    if (!q) return products.filter(p => p.stock > 0).slice(0, 24)
    return products.filter(p =>
      p.stock > 0 && (
        p.name.toLowerCase().includes(q) ||
        p.sku.toLowerCase().includes(q) ||
        p.barcode.toLowerCase().includes(q)
      )
    ).slice(0, 40)
  }, [products, search])

  const cartDetailed = cart.map(c => {
    const p = products.find(x => x.id === c.productId)!
    return { ...c, product: p }
  }).filter(c => c.product)

  const subtotal = cartDetailed.reduce((s, c) => s + c.product.sellPrice * c.qty, 0)
  const total = Math.max(0, subtotal - discount)
  const paid = payCash + payMpesa + payTigo + payBank + payCredit
  const change = paid - total

  const addToCart = (p: Product) => {
    setCart(prev => {
      const idx = prev.findIndex(c => c.productId === p.id)
      if (idx >= 0) {
        const next = [...prev]
        next[idx] = { ...next[idx], qty: next[idx].qty + 1 }
        return next
      }
      return [...prev, { productId: p.id, qty: 1, unit: p.unit }]
    })
  }

  const setQty = (productId: string, qty: number) => {
    if (qty <= 0) {
      setCart(cart.filter(c => c.productId !== productId))
    } else {
      setCart(cart.map(c => c.productId === productId ? { ...c, qty } : c))
    }
  }

  const clearCart = () => {
    setCart([])
    setDiscount(0)
    setCustomerId('')
    setWalkInName('')
    setPayCash(0); setPayMpesa(0); setPayTigo(0); setPayBank(0); setPayCredit(0)
  }

  const openCheckout = () => {
    if (cart.length === 0) return
    setPayCash(total)
    setPayMpesa(0); setPayTigo(0); setPayBank(0); setPayCredit(0)
    setCheckoutOpen(true)
  }

  const finalize = () => {
    if (paid < total) return
    const customer = customers.find(c => c.id === customerId)
    const customerName = customer?.name || walkInName || 'Mteja wa Kawaida'

    const items: SaleItem[] = cartDetailed.map(c => ({
      productId: c.product.id,
      productName: c.product.name,
      qty: c.qty,
      unit: c.product.unit,
      price: c.product.sellPrice,
      buyPrice: c.product.buyPrice,
    }))

    const payments: PaymentSplit[] = []

export default function ProfitLossPage() {
  const { sales, expenses, products } = useData()
  const [from, setFrom] = useState(addDays(today(), -30))
  const [to, setTo] = useState(today())

  const salesRange = sales.filter(s => s.createdAt.slice(0, 10) >= from && s.createdAt.slice(0, 10) <= to)
  const revenue = salesRange.reduce((s, x) => s + x.total, 0)
  const cogs = salesRange.reduce((s, x) => s + x.items.reduce((y, i) => y + i.buyPrice * i.qty, 0), 0)
  const grossProfit = revenue - cogs
  const opex = expenses.filter(e => e.status === 'approved' && e.date >= from && e.date <= to)
    .reduce((s, e) => s + e.amount, 0)
  const netProfit = grossProfit - opex
  const zakat = netProfit * 0.1
  const sadaka = netProfit * 0.1
  const yourShare = netProfit - zakat - sadaka

  // product breakdown
  const productSales: Record<string, { name: string; qty: number; revenue: number; profit: number }> = {}
  salesRange.forEach(s => s.items.forEach(i => {
    if (!productSales[i.productId]) productSales[i.productId] = { name: i.productName, qty: 0, revenue: 0, profit: 0 }
    productSales[i.productId].qty += i.qty
    productSales[i.productId].revenue += i.price * i.qty
    productSales[i.productId].profit += (i.price - i.buyPrice) * i.qty
  }))

          <label className="block text-xs text-slate-500 mb-1">Kuanzia</label>

            className="border rounded px-3 py-2 text-sm" />
        </div>
      </div>

        <Card className="p-5">
          <div className="text-xs text-slate-500 uppercase tracking-wide mb-3">Mchanganuo</div>

            <div className="flex justify-between"><span>Mauzo:</span><span className="font-medium">{fmtTsh(revenue)}</span></div>
            <div className="flex justify-between"><span>Gharama ya Bidhaa (COGS):</span><span className="text-slate-600">−{fmtTsh(cogs)}</span></div>
            <div className="flex justify-between border-t pt-2 font-semibold"><span>Faida Ghafi:</span><span>{fmtTsh(grossProfit)}</span></div>
            <div className="flex justify-between"><span>Gharama za Uendeshaji:</span><span className="text-red-600">−{fmtTsh(opex)}</span></div>

              <span>Faida Halisi:</span><span>{fmtTsh(netProfit)}</span>

  const slow = products.filter(p => productMoves[p.id]?.qty30 === 0 && p.stock > 0)

export default function UsersPage() {
  const { users, setUsers } = useAuth()
  const [open, setOpen] = useState(false)
  const [form, setForm] = useState(empty)
  const [delId, setDelId] = useState<string | null>(null)

  const save = () => {
    if (!form.name.trim() || !form.username.trim() || !form.password.trim()) return
    setUsers([...users, { ...form, id: uid() }])
    setOpen(false); setForm(empty)
  }

  const toggle = (id: string) => {
    setUsers(users.map(u => u.id === id ? { ...u, active: !u.active } : u))
  }

  const confirmDelete = () => {
    if (delId) { setUsers(users.filter(u => u.id !== delId)); setDelId(null) }
  }

  return (
    <Layout title="Usimamizi wa Watumiaji">
      <div className="flex justify-end mb-5">
        <Button onClick={() => { setForm(empty); setOpen(true) }}>

        </Button>
      </div>

      <Card>
        <Table head={['Jina', 'Username', 'Wajibu', 'Hali', 'Vitendo']}>

                  <span className="font-medium">{u.name}</span>
                </div>
              </Td>
              <Td className="font-mono text-xs">{u.username}</Td>
              <Td>
                <Badge color={
                  u.role === 'admin' ? 'red' : u.role === 'owner' ? 'purple'
                    : u.role === 'warehouse' ? 'amber' : 'blue'
                }>{u.role}</Badge>
              </Td>
              <Td>
                {u.active
                  ? <Badge color="green"><span className="flex items-center gap-1"><UserCheck size={10} /> Active</span></Badge>
                  : <Badge color="red"><span className="flex items-center gap-1"><UserX size={10} /> Disabled</span></Badge>}
              </Td>
              <Td>
                <div className="flex gap-2">
                  <button onClick={() => toggle(u.id)} className="text-xs text-brand-600 hover:underline">
                    {u.active ? 'Zima' : 'Washa'}
                  </button>
                  <button onClick={() => setDelId(u.id)} className="text-red-600 hover:bg-red-50 p-1 rounded">
                    <Trash2 size={14} />
                  </button>
                </div>
              </Td>
            </tr>
          ))}
        </Table>
      </Card>

      <Modal open={open} onClose={() => setOpen(false)} title="Mtumiaji Mpya" size="md">
        <div className="space-y-3">
          <Input label="Jina Kamili *" value={form.name} onChange={e => setForm({ ...form, name: e.target.value })} />

          <Input label="Password *" type="password" value={form.password} onChange={e => setForm({ ...form, password: e.target.value })} />

export function load<T>(key: string, fallback: T): T {
  try {
    const raw = localStorage.getItem(key)
    if (raw === null) return fallback
    return JSON.parse(raw) as T
  } catch (err) {
    console.error(`[storage] load failed for "${key}":`, err)
    localStorage.removeItem(key) // remove corrupt entry so it can't break again
    return fallback
  }
}

export function save<T>(key: string, value: T) {
  try {
    localStorage.setItem(key, JSON.stringify(value))
  } catch (err) {
    console.error(`[storage] save failed for "${key}":`, err)
  }
}

export const daysBetween = (a: string, b: string) =>
  Math.floor((new Date(b).getTime() - new Date(a).getTime()) / 86400000)

export const addDays = (date: string, days: number) => {
  const d = new Date(date)
  d.setDate(d.getDate() + days)
  return d.toISOString().slice(0, 10)
}
