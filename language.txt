// LANGUAGE CONTEXT
// Inasimamia lugha ya app (sw/en) na inatoa function ya t()
// ============================================================

import {
  createContext,
  useContext,
  useState,
  useCallback,
  useEffect,
  useMemo,
  type ReactNode,
} from 'react'
import { translations, LANGUAGES, type Language } from './translations'

const STORAGE_KEY = 'smart-hardware-lang'

// ---------- Aina ya Context ----------
interface LanguageContextValue {
  /** Lugha ya sasa (sw au en) */
  lang: Language
  /** Badilisha lugha */
  setLang: (l: Language) => void
  /**
   * Tafsiri key → maandishi
   * @example t('landing.login') → "Ingia" / "Login"
   * @example t('notif.count', { count: 5, unread: 2 }) → "5 taarifa - 2 hazijasomwa"
   */
  t: (key: string, vars?: Record<string, string | number>) => string
  /** Orodha ya lugha zinazopatikana */
  languages: typeof LANGUAGES
}

const LanguageContext = createContext<LanguageContextValue | undefined>(undefined)

// ============================================================
// PROVIDER
// ============================================================
export function LanguageProvider({ children }: { children: ReactNode }) {
  // ---------- 1. Pakia lugha iliyohifadhiwa ----------
  const [lang, setLangState] = useState<Language>(() => {
    try {
      const saved = localStorage.getItem(STORAGE_KEY) as Language | null
      if (saved && translations[saved]) return saved
    } catch {
      /* localStorage haipatikani (SSR au private mode) */
    }
    return 'sw'
  })

  // ---------- 2. Badilisha lugha ----------
  const setLang = useCallback((l: Language) => {
    setLangState(l)
    try {
      localStorage.setItem(STORAGE_KEY, l)
    } catch {
      /* ignore */
    }
    document.documentElement.lang = l
  }, [])

  // ---------- 3. Sasisha <html lang=""> ----------
  useEffect(() => {
    document.documentElement.lang = lang
  }, [lang])

  // ---------- 4. Function ya tafsiri ----------
  const t = useCallback(
    (key: string, vars?: Record<string, string | number>) => {
      // Tafuta kwenye lugha ya sasa, kisha sw (fallback), kisha key yenyewe
      let str = translations[lang][key] ?? translations.sw[key] ?? key

      // Badilisha {placeholders} kama zipo
      if (vars) {
        for (const [k, v] of Object.entries(vars)) {
          str = str.replace(new RegExp(`\\{${k}\\}`, 'g'), String(v))
        }
      }

      return str
    },
    [lang]
  )

  // ---------- 5. Context value ----------
  const value = useMemo(
    () => ({ lang, setLang, t, languages: LANGUAGES }),
    [lang, setLang, t]
  )

  return <LanguageContext.Provider value={value}>{children}</LanguageContext.Provider>
}

// ============================================================
// HOOK - Tumia kwenye components zote
// ============================================================
export function useLanguage() {
  const ctx = useContext(LanguageContext)
  if (!ctx) {
    throw new Error('useLanguage must be used within LanguageProvider')
  }
  return ctx
},

import { useState, useRef, useEffect } from 'react'
import { Globe, ChevronDown, Check } from 'lucide-react'
import { useLanguage } from '../i18n/LanguageContext'

export default function LanguageSwitcher() {
  const { lang, setLang, languages, t } = useLanguage()
  const [open, setOpen] = useState(false)
  const ref = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const handler = (e: MouseEvent) => {
      if (ref.current && !ref.current.contains(e.target as Node)) setOpen(false)
    }
    document.addEventListener('mousedown', handler)
    return () => document.removeEventListener('mousedown', handler)
  }, [])

  const current = languages.find((l) => l.key === lang)!

  return (
    <div className="relative" ref={ref}>
      <button
        onClick={() => setOpen((v) => !v)}
        className="inline-flex items-center gap-1.5 px-2.5 py-2 rounded-lg text-sm font-medium text-slate-700 hover:bg-slate-100 transition"
        title={t('common.language')}
      >
        <Globe size={16} />
        <span className="hidden sm:inline">{current.flag} {current.label}</span>
        <span className="sm:hidden">{current.flag}</span>
        <ChevronDown size={14} className={`transition ${open ? 'rotate-180' : ''}`} />
      </button>

      {open && (
        <div className="absolute right-0 mt-1 w-44 bg-white rounded-xl border border-slate-200 shadow-lg py-1 z-50">
          {languages.map((l) => (
            <button
              key={l.key}
              onClick={() => {
                setLang(l.key)
                setOpen(false)
              }}
              className={`w-full flex items-center gap-2 px-3 py-2 text-sm hover:bg-slate-50 ${
                lang === l.key ? 'text-brand-700 font-semibold' : 'text-slate-700'
              }`}
            >
              <span>{l.flag}</span>
              <span className="flex-1 text-left">{l.label}</span>
              {lang === l.key && <Check size={14} className="text-brand-600" />}
            </button>
          ))}
        </div>
      )}
    </div>
  )
},
